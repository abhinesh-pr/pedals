import 'dart:async';
import 'dart:ui' as ui;
import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:flutter/material.dart';
import 'package:flutter/services.dart' show rootBundle;
import 'package:get/get.dart';
import '../../services/cycle_services.dart';
import '../../viewmodels/cycle_slot_model.dart';
import 'package:pedals/views/users/user_profile.dart';

class UserDashboard extends StatefulWidget {
  final String? uemail;

  const UserDashboard({Key? key, required this.uemail}) : super(key: key);

  @override
  State<UserDashboard> createState() => _UserDashboard();
}

class _UserDashboard extends State<UserDashboard>
    with TickerProviderStateMixin {
  String username = '';
  String userId = '';
  int? selectedPinIndex;
  final CycleService _cycleService = CycleService();
  final Map<int, Stream<List<CycleSlot>>> pinStreams = {};
  ui.Image? _image;
  final TransformationController _transformationController =
      TransformationController();
  final StreamController<int> _pinSelectionController = StreamController<int>();

  final Map<String, Map<int, String>> cycleStatesByStand = {};

  final Map<int, String> pinStandMap = {
    0: "STAND_A",
    1: "STAND_B",
    2: "STAND_C",
    3: "STAND_D",
    4: "STAND_E",
  };

  final Map<int, String> standNames = {
    0: "Main Gate",
    1: "Admission Block",
    2: "KG CAS/ KG IIM",
    3: "IT Tower",
    4: "KiTE",
  };

  final List<Map<String, double>> pins = [
    {"x": 0.46, "y": 0.80},
    {"x": 0.53, "y": 0.67},
    {"x": 0.18, "y": 0.5},
    {"x": 0.2, "y": 0.4},
    {"x": 0.43, "y": 0.33},
  ];

  final List<String> pinLabels = ["A", "B", "C", "D", "E"];
  List<Map<String, dynamic>> demoSlots = [];

  void _initDemoCycleSlots() {
    final standId = pinStandMap[selectedPinIndex ?? 0] ?? '';

    if (!cycleStatesByStand.containsKey(standId)) {
      cycleStatesByStand[standId] = {};
      for (int i = 0; i < 2; i++) {
        cycleStatesByStand[standId]![i + 1] =
            'locked'; // Slots 1 and 2 are locked by default
      }
    }

    // Always have 3 slots per stand, with 2 cycles present and 1 empty slot
    demoSlots = List.generate(3, (index) {
      // Make the 3rd slot always empty for swapping
      final isEmpty = index == 2;
      final slotId = index + 1;

      // Get the current state for this slot if it exists
      final status =
          isEmpty ? 'empty' : cycleStatesByStand[standId]![slotId] ?? 'locked';

      return {
        'slotId': slotId,
        'cycleId':
            isEmpty
                ? ''
                : 'Cycle_${(selectedPinIndex ?? 0) * 100 + index + 101}',
        'status': status,
      };
    });
  }

  Future<void> fetchUserData(String email) async {
    try {
      final userSnapshot =
          await FirebaseFirestore.instance
              .collection('users')
              .where('email', isEqualTo: email)
              .limit(1)
              .get();

      if (userSnapshot.docs.isNotEmpty) {
        final userDocId = userSnapshot.docs.first.id;
        final userDoc =
            await FirebaseFirestore.instance
                .collection('users')
                .doc(userDocId)
                .get();

        if (userDoc.exists) {
          final data = userDoc.data()!;
          setState(() {
            username = data['fullName'] ?? '';
            userId = userDocId;
          });
        }
      }
    } catch (e) {
      print('Error fetching user data: $e');
    }
  }

  @override
  void initState() {
    super.initState();
    _loadImage('assets/images/map.jpg');
    fetchUserData(widget.uemail ?? '');

    pinStandMap.forEach((pinIndex, standId) {
      final stream = _cycleService.getRealtimeSlots(standId);
      pinStreams[pinIndex] = stream;
      stream.listen((cycles) {
        if (!mounted) return;
        // Handle updates if needed
      });
    });

    _pinSelectionController.stream.listen((index) {
      setState(() {
        // If the same pin is clicked again, close the card
        if (selectedPinIndex == index) {
          selectedPinIndex = null;
        } else {
          // Different pin clicked, update with the new stand details
          selectedPinIndex = index;
          _initDemoCycleSlots();
        }
      });
    });
  }

  @override
  void dispose() {
    _pinSelectionController.close();
    _transformationController.dispose();
    super.dispose();
  }

  Future<void> _loadImage(String path) async {
    final data = await rootBundle.load(path);
    final image = await decodeImageFromList(data.buffer.asUint8List());
    if (!mounted) return;
    setState(() => _image = image);
  }

  void _resetMapToInitialView() {
    _transformationController.value = Matrix4.identity();
  }

  List<CycleSlot> _getDemoCycleSlots() {
    // Convert demoSlots (Map<String, dynamic>) to a List<CycleSlot>
    return demoSlots.map((slotData) {
      return CycleSlot(
        slotId: slotData['slotId'],
        cycleIdApp: slotData['cycleId'] ?? '',
        cycleStatus: slotData['status'] ?? '',
        standId: pinStandMap[selectedPinIndex ?? 0] ?? '',
        cycleId: '',
        standStatus: '',
        workId: '',
      );
    }).toList();
  }

  Widget _buildStatusDetails(String status) {
    Color statusColor;
    IconData statusIcon;

    switch (status.toLowerCase()) {
      case 'under review':
        statusColor = Colors.orange;
        statusIcon = Icons.hourglass_top;
        break;
      case 'in progress':
        statusColor = Colors.blue;
        statusIcon = Icons.engineering;
        break;
      case 'resolved':
        statusColor = Colors.green;
        statusIcon = Icons.check_circle;
        break;
      default:
        statusColor = Colors.grey;
        statusIcon = Icons.info;
    }

    return Container(
      padding: const EdgeInsets.all(5),
      decoration: BoxDecoration(
        color: statusColor.withOpacity(0.1),
        borderRadius: BorderRadius.circular(12),
        border: Border.all(color: statusColor.withOpacity(0.5)),
      ),
      child: Row(
        children: [
          Icon(statusIcon, color: statusColor, size: 24),
          const SizedBox(width: 12),
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(
                  status,
                  style: TextStyle(
                    fontWeight: FontWeight.bold,
                    fontSize: 16,
                    color: statusColor,
                  ),
                ),
                const SizedBox(height: 4),
                Text(
                  status == 'Under Review'
                      ? 'Your complaint is currently being reviewed by our team.'
                      : 'Status description goes here.',
                  style: TextStyle(fontSize: 14, color: Colors.grey[700]),
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }

  // This function builds the bottom sheet style UI that appears below the map
  Widget _buildStandInfoPanel(int pinIndex) {
    final standId = pinStandMap[pinIndex] ?? '';
    final slots = _getDemoCycleSlots();
    final standName = standNames[pinIndex] ?? 'Unknown Location';

    return SingleChildScrollView(
      child: GestureDetector(
        // Prevent taps on the card from closing it (due to the outer gesture detector)
        onTap: () {},
        onVerticalDragEnd: (details) {
          // If the drag ends with an upward velocity, close the panel
          if (details.primaryVelocity != null &&
              details.primaryVelocity! > 300) {
            setState(() {
              selectedPinIndex = null;
            });
          }
        },
        child: Container(
          width: double.maxFinite,
          // Full width for the card
          margin: const EdgeInsets.only(top: 30),
          padding: const EdgeInsets.all(16),
          decoration: BoxDecoration(
            color: Colors.white,
            borderRadius: BorderRadius.circular(20),
            boxShadow: [
              BoxShadow(
                color: Colors.black.withOpacity(0.1),
                blurRadius: 6,
                offset: const Offset(0, 3),
              ),
            ],
          ),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            mainAxisSize: MainAxisSize.max,
            children: [
              // Handle at top like a bottom sheet - made larger for easier grabbing
              Center(
                child: Container(
                  width: 50,
                  height: 5,
                  decoration: BoxDecoration(
                    color: Colors.grey[300],
                    borderRadius: BorderRadius.circular(10),
                  ),
                ),
              ),
              Text(
                standName,
                style: const TextStyle(
                  fontWeight: FontWeight.bold,
                  fontSize: 18,
                  color: Colors.black,
                ),
              ),
              const SizedBox(height: 7),
              Text(
                'Stand ID: $standId (Location ${pinLabels[pinIndex]})',
                style: TextStyle(fontSize: 14, color: Colors.grey[700]),
              ),
              const SizedBox(height: 16),
              const Text(
                'Available Cycles',
                style: TextStyle(
                  fontSize: 18,
                  fontWeight: FontWeight.bold,
                  color: Colors.black,
                ),
              ),
              const SizedBox(height: 12),

              // Empty slots - show all slots, including empty ones
              ...slots.map((cycle) {
                final isLocked = cycle.cycleStatus.toLowerCase() == "locked";
                final isEmpty = cycle.cycleIdApp.isEmpty;

                return Card(
                  elevation: 4,
                  margin: const EdgeInsets.symmetric(vertical: 6),
                  shape: RoundedRectangleBorder(
                    borderRadius: BorderRadius.circular(12),
                  ),
                  child: ListTile(
                    leading: Icon(
                      isEmpty
                          ? Icons.indeterminate_check_box_outlined
                          : Icons.pedal_bike,
                      color: isEmpty ? Colors.grey : Colors.blue,
                    ),
                    title: Text(
                      isEmpty
                          ? "Empty Slot"
                          : cycle.cycleIdApp.replaceAll('_', ' '),
                    ),
                    subtitle: Text("Slot : ${cycle.slotId}"),
                    trailing:
                        isEmpty
                            ? const Text(
                              "Available",
                              style: TextStyle(
                                color: Colors.grey,
                                fontWeight: FontWeight.bold,
                              ),
                            )
                            : ElevatedButton.icon(
                              icon: Icon(
                                isLocked ? Icons.lock : Icons.lock_open,
                              ),
                              label: Text(isLocked ? "Unlock" : "Unlocked"),
                              style: ElevatedButton.styleFrom(
                                backgroundColor:
                                    isLocked ? Colors.blue : Colors.green,
                              ),
                              onPressed: () {
                                if (isLocked) {
                                  // Show a snackbar immediately to indicate action
                                  ScaffoldMessenger.of(context).showSnackBar(
                                    const SnackBar(
                                      content: Text("Unlocking cycle..."),
                                      duration: Duration(seconds: 1),
                                    ),
                                  );

                                  // Update the global state to persist unlock status
                                  setState(() {
                                    final standId = pinStandMap[pinIndex];
                                    if (standId != null) {
                                      if (!cycleStatesByStand.containsKey(
                                        standId,
                                      )) {
                                        cycleStatesByStand[standId] = {};
                                      }
                                      cycleStatesByStand[standId]![cycle
                                              .slotId] =
                                          'unlocked';

                                      // Update the local demoSlots for immediate UI feedback
                                      for (
                                        var i = 0;
                                        i < demoSlots.length;
                                        i++
                                      ) {
                                        if (demoSlots[i]['slotId'] ==
                                            cycle.slotId) {
                                          demoSlots[i]['status'] = 'unlocked';
                                          break;
                                        }
                                      }
                                    }
                                  });

                                  // Then call the service
                                  _cycleService.unlockCycle(
                                    standId: cycle.standId,
                                    slotId: cycle.slotId,
                                  );
                                } else {
                                  ScaffoldMessenger.of(context).showSnackBar(
                                    const SnackBar(
                                      content: Text(
                                        "You cannot lock the cycle through app for now",
                                      ),
                                      duration: Duration(seconds: 2),
                                    ),
                                  );
                                }
                              },
                            ),
                  ),
                );
              }).toList(),
            ],
          ),
        ),
      ),
    );
  }

  List<Widget> _buildSlotCard(List<CycleSlot> cycles) {
    final filteredCycles =
        cycles.where((cycle) => cycle.cycleIdApp.isNotEmpty).toList();
    filteredCycles.sort((a, b) => a.slotId.compareTo(b.slotId));

    return filteredCycles.map((cycle) {
      final isLocked = cycle.cycleStatus.toLowerCase() == "locked";

      return Card(
        elevation: 1,
        margin: const EdgeInsets.symmetric(vertical: 20),
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
        child: ListTile(
          leading: const Icon(Icons.pedal_bike),
          title: Text(cycle.cycleIdApp.replaceAll('_', ' ')),
          subtitle: Text("Slot : ${cycle.slotId}"),
          trailing: ElevatedButton.icon(
            icon: Icon(isLocked ? Icons.lock : Icons.lock_open),
            label: Text(isLocked ? "Locked" : "Unlocked"),
            style: ElevatedButton.styleFrom(
              backgroundColor: isLocked ? Colors.redAccent : Colors.green,
            ),
            onPressed: () {
              if (isLocked) {
                _cycleService.unlockCycle(
                  standId: cycle.standId,
                  slotId: cycle.slotId,
                );
              } else {
                ScaffoldMessenger.of(context).showSnackBar(
                  const SnackBar(
                    content: Text(
                      "You cannot lock the cycle through app for now",
                    ),
                    duration: Duration(seconds: 2),
                  ),
                );
              }
            },
          ),
        ),
      );
    }).toList();
  }

  @override
  Widget build(BuildContext context) {
    if (_image == null) {
      return const Scaffold(body: Center(child: CircularProgressIndicator()));
    }

    // Add a tap detector outside the map to close the card when clicking elsewhere
    void closeCardIfTappedOutside(TapDownDetails details) {
      if (selectedPinIndex != null) {
        setState(() {
          selectedPinIndex = null;
        });
      }
    }

    final aspectRatio = _image!.width / _image!.height;

    return Scaffold(
      appBar: AppBar(
        title: const Text(
          "Pedals",
          style: TextStyle(
            color: Colors.blue,
            fontSize: 20,
            fontWeight: FontWeight.bold,
          ),
        ),
        actions: [
          IconButton(
            icon: const Icon(
              Icons.account_circle_outlined,
              size: 28,
              color: Colors.black,
            ),
            onPressed: () => Get.to(() => ProfilePage()),
          ),
        ],
        elevation: 0,
        backgroundColor: Colors.white,
        bottom: PreferredSize(
          preferredSize: const Size.fromHeight(1.0),
          child: Container(color: const Color(0xFFCFCFCF), height: 0.7),
        ),
      ),
      body: LayoutBuilder(
        builder: (context, constraints) {
          final containerWidth = constraints.maxWidth;
          final containerHeight = constraints.maxHeight;
          const double horizontalPadding = 2.0;
          final availableWidth = containerWidth - (2 * horizontalPadding);

          // Calculate the minimum map size - scaled down when card is visible
          // Scale the map down more aggressively when a pin is selected
          final double mapScale = selectedPinIndex != null ? 0.65 : 1.0;
          final double mapWidth = availableWidth * mapScale;
          final double mapHeight = mapWidth / aspectRatio;

          // Determine vertical positioning
          final double mapTopPosition = selectedPinIndex != null ? 10.0 : 20.0;

          return GestureDetector(
            // Close the card when tapping outside of map and card
            behavior: HitTestBehavior.translucent,
            onTapDown: closeCardIfTappedOutside,
            child: Padding(
              padding: const EdgeInsets.symmetric(
                horizontal: horizontalPadding,
              ),
              child: Column(
                children: [
                  AnimatedContainer(
                    duration: const Duration(milliseconds: 300),
                    height: mapHeight + mapTopPosition,
                    alignment: Alignment.topCenter,
                    child: AnimatedPadding(
                      duration: const Duration(milliseconds: 300),
                      padding: EdgeInsets.only(top: mapTopPosition),
                      child: ClipRRect(
                        borderRadius: BorderRadius.circular(20),
                        child: InteractiveViewer(
                          transformationController: _transformationController,
                          minScale: 1.0,
                          maxScale: 5.0,
                          panEnabled: true,
                          clipBehavior: Clip.hardEdge,
                          boundaryMargin: const EdgeInsets.all(0),
                          child: SizedBox(
                            width: mapWidth,
                            height: mapHeight,
                            child: Stack(
                              children: [
                                // Map Image
                                Image.asset(
                                  'assets/images/map.jpg',
                                  width: mapWidth,
                                  height: mapHeight,
                                  fit: BoxFit.fill,
                                ),
                                // Location Pins
                                ...pins.asMap().entries.map((entry) {
                                  int index = entry.key;
                                  final pin = entry.value;
                                  final left = pin["x"]! * mapWidth;
                                  final top = pin["y"]! * mapHeight;
                                  return Positioned(
                                    left: left - 18,
                                    top: top - 36,
                                    child: GestureDetector(
                                      onTap: () {
                                        // Stop propagation to prevent the outer gesture detector from triggering
                                        _pinSelectionController.add(index);
                                      },
                                      child: Column(
                                        children: [
                                          const Icon(
                                            Icons.location_on,
                                            color: Colors.red,
                                            size: 36,
                                          ),
                                          const SizedBox(height: 1),
                                          Container(
                                            padding: const EdgeInsets.symmetric(
                                              horizontal: 6,
                                              vertical: 2,
                                            ),
                                            decoration: BoxDecoration(
                                              color: Colors.white.withOpacity(
                                                0.5,
                                              ),
                                              borderRadius:
                                                  BorderRadius.circular(6),
                                            ),
                                            child: Text(
                                              pinLabels[index],
                                              style: const TextStyle(
                                                color: Colors.black,
                                                fontSize: 10,
                                                fontWeight: FontWeight.bold,
                                              ),
                                            ),
                                          ),
                                        ],
                                      ),
                                    ),
                                  );
                                }).toList(),
                              ],
                            ),
                          ),
                        ),
                      ),
                    ),
                  ),

                  // Stand Info Panel - Full width, independent of map scaling
                  Expanded(
                    child: AnimatedOpacity(
                      duration: const Duration(milliseconds: 300),
                      opacity: selectedPinIndex != null ? 1.0 : 0.0,
                      child:
                          selectedPinIndex != null
                              ? _buildStandInfoPanel(selectedPinIndex!)
                              : const SizedBox(),
                    ),
                  ),
                ],
              ),
            ),
          );
        },
      ),
    );
  }
}
